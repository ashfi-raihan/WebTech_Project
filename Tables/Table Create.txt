CREATE DATABASE skillbridge_db; 
USE skillbridge_db; 
CREATE TABLE users ( id INT AUTO_INCREMENT PRIMARY KEY, full_name VARCHAR(100), email VARCHAR(100) UNIQUE, password VARCHAR(255), role VARCHAR(50), created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ); 
CREATE TABLE skills ( skill_id INT AUTO_INCREMENT PRIMARY KEY, user_id INT, title VARCHAR(255), category VARCHAR(100), description TEXT, mode VARCHAR(50), price DECIMAL(10,2), created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ); 
CREATE TABLE bookings ( booking_id INT AUTO_INCREMENT PRIMARY KEY, skill_id INT, learner_id INT, booking_date DATE, booking_time VARCHAR(50), notes TEXT, status VARCHAR(20) DEFAULT 'Pending', created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ); 
CREATE TABLE reviews ( review_id INT AUTO_INCREMENT PRIMARY KEY, booking_id INT, rating INT, feedback TEXT, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP );
